## ids.py

```python
import sqlite3,glob,sys,os
out=[]
for f in sorted(glob.glob('apps/mobile/assets/content/tresc-*.db')):
    j=f.rsplit('-',1)[1][:-3]; c=sqlite3.connect(f)
    for t,k in [('modules','id'),('lessons','id'),('drills','id'),('exam_drills','id'),('rules','id'),('scenarios','id'),('ranges','id'),('terms','key'),('numbers','key')]:
        for (v,) in c.execute(f'select {k} from {t}'): out.append(f'{j}\t{t}\t{v}')
open(sys.argv[1],'w').write('\n'.join(out)+'\n'); print(len(out),'wierszy', len(glob.glob('apps/mobile/assets/content/tresc-*.db')),'baz')
```
