# --- skrypt ---
import { handEquity, parseCard } from '/home/user/Szkola_pokera/packages/poker-core/src/index.ts';
const R = '23456789TJQKA';
const konf = [['h','c','s','d'],['h','s','s','d'],['h','c','s','h'],['h','s','s','h'],['h','c','s','c']];
const out: string[] = [];
for (let x = 14; x >= 10; x--) for (let k1 = 2; k1 <= x - 2; k1++) for (const k2 of [k1 + 1]) for (const [a, b, c, d] of konf) {
  const h1 = [parseCard(R[x-2]+a), parseCard(R[k1-2]+b)], h2 = [parseCard(R[x-2]+c), parseCard(R[k2-2]+d)];
  const e = handEquity([h1 as any, h2 as any], [], { exactLimit: 2_000_000 });
  out.push(`${e.exact ? 'dokł' : 'MC'} ${R[x-2]}${a}${R[k1-2]}${b} vs ${R[x-2]}${c}${R[k2-2]}${d} eq=${e.equity[0]!.toFixed(4)} win=${e.win[0]!.toFixed(4)} lose=${e.win[1]!.toFixed(4)}`);
}
out.sort((p, q) => Number(q.split('eq=')[1]!.slice(0, 6)) - Number(p.split('eq=')[1]!.slice(0, 6)));
console.log(out.slice(0, 8).join('\n'));
