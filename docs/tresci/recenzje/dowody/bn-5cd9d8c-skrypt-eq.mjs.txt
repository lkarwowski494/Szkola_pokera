// Dokładne equity (pełny przegląd C(48,5) boardów) ręki zdominowanej vs dominującej. Karta = ranga*4+kolor, ranga 0..12 (2..A).
function ev7(c) { // zwraca liczbę porównywalną: kategoria*13^5 + kickery
  const rc = new Array(13).fill(0), sc = [0,0,0,0], sr = [0,0,0,0];
  for (const x of c) { const r = x >> 2, s = x & 3; rc[r]++; sc[s]++; sr[s] |= 1 << r; }
  const val = (cat, ks) => { let v = cat; for (let i = 0; i < 5; i++) v = v * 13 + (ks[i] ?? 0); return v; };
  const straightTop = (mask) => { const m = mask | ((mask >> 12) & 1 ? 1 << -1 >>> 0 : 0); for (let t = 12; t >= 4; t--) { let ok = true; for (let k = 0; k < 5; k++) if (!((mask >> (t - k)) & 1)) { ok = false; break; } if (ok) return t; } if ((mask & 0b1000000001111) === 0b1000000001111) return 3; return -1; };
  for (let s = 0; s < 4; s++) if (sc[s] >= 5) { const st = straightTop(sr[s]); if (st >= 0) return val(8, [st]); }
  const quads = [], trips = [], pairs = [], singles = [];
  for (let r = 12; r >= 0; r--) { if (rc[r] === 4) quads.push(r); else if (rc[r] === 3) trips.push(r); else if (rc[r] === 2) pairs.push(r); else if (rc[r] === 1) singles.push(r); }
  if (quads.length) { const k = [...trips, ...pairs, ...singles].sort((a,b)=>b-a)[0]; return val(7, [quads[0], k]); }
  if (trips.length && (trips.length > 1 || pairs.length)) { const p = [trips[1] ?? -1, pairs[0] ?? -1].sort((a,b)=>b-a)[0]; return val(6, [trips[0], p]); }
  for (let s = 0; s < 4; s++) if (sc[s] >= 5) { const ks = []; for (let r = 12; r >= 0 && ks.length < 5; r--) if ((sr[s] >> r) & 1) ks.push(r); return val(5, ks); }
  let mask = 0; for (let r = 0; r < 13; r++) if (rc[r]) mask |= 1 << r; const st = straightTop(mask); if (st >= 0) return val(4, [st]);
  if (trips.length) return val(3, [trips[0], ...[...pairs, ...singles].sort((a,b)=>b-a).slice(0,2)]);
  if (pairs.length >= 2) { const rest = [...pairs.slice(2), ...singles].sort((a,b)=>b-a); return val(2, [pairs[0], pairs[1], rest[0]]); }
  if (pairs.length) return val(1, [pairs[0], ...singles.slice(0,3)]);
  return val(0, singles.slice(0,5));
}
const R = '23456789TJQKA', S = 'shdc';
const card = (s) => R.indexOf(s[0]) * 4 + S.indexOf(s[1]);
function equity(h1, h2) {
  const a = h1.map(card), b = h2.map(card), used = new Set([...a, ...b]); const d = []; for (let i = 0; i < 52; i++) if (!used.has(i)) d.push(i);
  let w = 0, t = 0, n = 0; const bd = new Array(5);
  for (let i = 0; i < 48; i++) { bd[0] = d[i]; for (let j = i+1; j < 48; j++) { bd[1] = d[j]; for (let k = j+1; k < 48; k++) { bd[2] = d[k]; for (let l = k+1; l < 48; l++) { bd[3] = d[l]; for (let m = l+1; m < 48; m++) { bd[4] = d[m];
    const x = ev7([...a, ...bd]), y = ev7([...b, ...bd]); n++; if (x > y) w++; else if (x === y) t++; } } } } }
  return ((w + t / 2) / n);
}
// sanity: AA vs KK ok. 0,82; AKo vs AQo ok. 0,74
const pary = process.argv.slice(2).map((x) => x.split(':'));
for (const [h1, h2] of pary) { const e = equity([h1.slice(0,2), h1.slice(2,4)], [h2.slice(0,2), h2.slice(2,4)]); console.log(`${h1} vs ${h2}: ${e.toFixed(4)}`); }
