// Sonda DI (reaudit 06-tresci-001): wyrocznia niezależna od testu autora, pisana od zera.
import { join, resolve } from 'node:path';
import { writeFileSync } from 'node:fs';
import { it } from 'vitest';
import { FULL_DECK, generateVariant, rankOf, suitOf, type Card, type GeneratorTemplate, type ScenarioLevel } from '@szkola/poker-core';
import { loadScenarioTemplates } from '../src/scenariusze';
import { number } from './kit';

const root = resolve(import.meta.dirname, '../../..');
const pl = loadScenarioTemplates(join(root, 'content', 'pl', 'scenarios')).map((x) => x.template);
const ZD = '{{n:pair.outs.per-rank}}';
// Ręka pięciokartowa najlepsza z 7: wystarczy sprawdzić strit i kolor (para/set nie zmienia „outu drawa”).
function maStrita(k: readonly Card[]) {
  const r = new Set<number>(); for (const c of k) { const x = rankOf(c) + 2; r.add(x); if (x === 14) r.add(1); }
  for (let lo = 1; lo <= 10; lo++) { let ok = true; for (let j = 0; j < 5; j++) if (!r.has(lo + j)) ok = false; if (ok) return true; }
  return false;
}
function maKolor(k: readonly Card[]) { const n = [0, 0, 0, 0]; for (const c of k) n[suitOf(c)]++; return n.some((x) => x >= 5); }
it('sonda', () => {
  const out: string[] = [];
  for (const id of ['sc-01', 'sc-02', 'sc-03', 'sc-19']) {
    const t = pl.find((x) => x.id === id)!;
    const gen = { id: t.id, stackBb: t.stackBb, positions: t.positions, levels: t.bounds, decisions: t.decisions } as GeneratorTemplate;
    const rozklad = new Map<string, number>(); const outyDrawa = new Map<string, number>(); let pary = 0, punkty = 0, zNadkarta = 0; const przyklady: string[] = [];
    for (const level of t.levels as ScenarioLevel[]) for (let seed = 1; seed <= 2000; seed++) {
      const v = generateVariant(gen, level, seed);
      v.decisions.forEach((d, i) => {
        const ex = ((t.decisions[i] as any).actions ?? []).map((a: any) => a.explanation as string);
        if (!ex.some((e: string) => e.includes(ZD))) return;
        punkty++;
        const stol = v.board.slice(0, d.street === 'flop' ? 3 : 4);
        const widoczne = new Set<Card>([...v.hero, ...stol]);
        const reszta = FULL_DECK.filter((c) => !widoczne.has(c));
        const outy = reszta.filter((c) => maStrita([...v.hero, ...stol, c]) || maKolor([...v.hero, ...stol, c]));
        const odNum = ex.map((e: string) => e.match(/\{\{n:(outs\.[a-z]+)\}\}/)?.[1]).find(Boolean);
        const kl = `${d.street} ${odNum ?? '-'}=${odNum ? number(odNum) : '-'} real=${outy.length}`; outyDrawa.set(kl, (outyDrawa.get(kl) ?? 0) + 1);
        if (v.hero[0] !== undefined && rankOf(v.hero[0]) === rankOf(v.hero[1])) pary++;
        const top = Math.max(...stol.map(rankOf));
        const rangi = [...new Set(v.hero.map(rankOf).filter((r) => r > top))];
        if (rangi.length) zNadkarta++;
        for (const r of rangi) {
          const nowe = reszta.filter((c) => rankOf(c) === r && !outy.includes(c)).length;
          const k = `${d.street} nowe=${nowe}`; rozklad.set(k, (rozklad.get(k) ?? 0) + 1);
          if (nowe !== 3 && przyklady.length < 3) przyklady.push(`${level}/${seed}/p${i + 1} ${v.hero.join(',')} | ${stol.join(',')} r${r} nowe=${nowe}`);
        }
      });
    }
    out.push(`${id}: punkty ze zdaniem ${punkty}, z nadkartą ${zNadkarta}, pary kieszonkowe ${pary}`);
    out.push(`  nadkarty: ${[...rozklad].sort().map(([k, n]) => `${k}:${n}`).join('  ')}`);
    out.push(`  outy drawa (liczba w zdaniu vs z kart): ${[...outyDrawa].sort().map(([k, n]) => `${k}:${n}`).join('  ')}`);
    for (const p of przyklady) out.push(`  przykład ${p}`);
  }
  writeFileSync(process.env.SONDA_OUT!, out.join('\n') + '\n');
}, 600_000);
