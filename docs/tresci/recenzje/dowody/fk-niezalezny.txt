import itertools
R='23456789TJQKA'; S='shdc'
DECK=[r+s for r in R for s in S]
def rank5(c):
    rs=sorted((R.index(x[0]) for x in c),reverse=True); su=[x[1] for x in c]
    cnt={r:rs.count(r) for r in set(rs)}; grp=sorted(cnt.items(),key=lambda x:(-x[1],-x[0]))
    fl=len(set(su))==1; u=sorted(set(rs),reverse=True); st=None
    if len(u)==5 and u[0]-u[4]==4: st=u[0]
    if u==[12,3,2,1,0]: st=3
    if st is not None and fl: return (8,st)
    if grp[0][1]==4: return (7,grp[0][0],grp[1][0])
    if grp[0][1]==3 and grp[1][1]==2: return (6,grp[0][0],grp[1][0])
    if fl: return (5,*rs)
    if st is not None: return (4,st)
    if grp[0][1]==3: return (3,grp[0][0],*[g for g,_ in grp[1:]])
    if grp[0][1]==2 and grp[1][1]==2: return (2,grp[0][0],grp[1][0],grp[2][0])
    if grp[0][1]==2: return (1,grp[0][0],*[g for g,_ in grp[1:]])
    return (0,*rs)
def best(c): return max(rank5(x) for x in itertools.combinations(c,5))
def combos(hc):
    if len(hc)==2: return [ (hc[0]+a,hc[1]+b) for a,b in itertools.combinations(S,2)]
    a,b,t=hc[0],hc[1],hc[2]
    if t=='s': return [(a+s,b+s) for s in S]
    return [(a+x,b+y) for x in S for y in S if x!=y]
def flop_vec(hero,vil,board):
    known=set(hero+vil+board); unseen=[c for c in DECK if c not in known]
    win=[c for c in unseen if best(vil+board+[c])>=best(hero+board+[c])]
    return win,len(unseen)

def wl(calls,folds):
    dead=set(hero+board); W=L=T=0; fb=0; fn=0
    for hc in calls:
        for c in combos(hc):
