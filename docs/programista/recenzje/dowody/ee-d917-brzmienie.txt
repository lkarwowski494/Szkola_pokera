import { execSync } from 'node:child_process';
import { parse } from 'yaml';
const R = '/home/user/Szkola_pokera';
const pliki = execSync('git ls-files "content/*/scenarios/*.yaml"', { cwd: R }).toString().trim().split('\n');
let por = 0, zbl = 0, roz = [], zmienionePliki = 0, bloki = 0, zlyKsztalt = [];
for (const f of pliki) {
  let stary; try { stary = execSync(`git show f86c72fc:${f}`, { cwd: R, stdio: ['pipe','pipe','ignore'] }).toString(); } catch { roz.push(`${f}: brak w f86c72fc`); continue; }
  const nowy = execSync(`git show 539919af:${f}`, { cwd: R }).toString();
  if (stary !== nowy) zmienionePliki++;
  const A = parse(stary), B = parse(nowy);
  // reszta pliku poza wyjaśnieniami bez zmian
  const bezWyj = (t) => JSON.stringify(t, (k, v) => (k === 'explanation' ? undefined : v));

  B.decisions.forEach((d, i) => d.actions.forEach((a, j) => {
    por++;
