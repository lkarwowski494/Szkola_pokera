  ['M2 list-hand: dopuszczalna jako błąd', WAR, "d.acceptable.includes(a) ? 'dopuszczalne' : 'blad'", "d.acceptable.includes(a) ? 'blad' : 'blad'", 'mobile', FG],
  ['M3 list-hand: każda niepoprawna dopuszczalna', WAR, "d.acceptable.includes(a) ? 'dopuszczalne' : 'blad'", "'dopuszczalne'", 'mobile', FG],
  ['M4 river-catch/defend: druga akcja niedokładność zamiast błędu', WAR, "return a === poprawnaAkcja(d) ? 'zgodne' : 'blad';\n}", "return a === poprawnaAkcja(d) ? 'zgodne' : 'niedokladnosc';\n}", 'mobile', FG],
  ['M5 river-catch/defend: poprawna zawsze call', WAR, "if (p.point === 'pick-hand' || p.point === 'turn-bet' || p.point === 'river-catch' || p.point === 'defend') return p.correct;", "if (p.point === 'river-catch' || p.point === 'defend') return 'call';\n  if (p.point === 'pick-hand' || p.point === 'turn-bet') return p.correct;", 'mobile', FG],

  ['M12 bramka: range-spot play przepuszczony', SCN, "if (d.kind === 'range-spot' && d.play) {", "if (false) {", 'cb', 'test/'],

const tylko = process.argv[3];
for (const [nazwa, plik, z, na, pakiet, wzor] of M) {
  if (tylko && !nazwa.startsWith(tylko + ' ')) continue;
  const org = readFileSync(plik, 'utf8');
  if (org.split(z).length !== 2) { console.log(`${nazwa}: WZORZEC NIE WYSTĘPUJE RAZ (${org.split(z).length - 1})`); continue; }
  writeFileSync(plik, org.replace(z, na));
  const cmd = pakiet === 'mobile' ? `cd ${W}/apps/mobile && npx jest --ci ${wzor}` : `cd ${W}/tools/content-build && npx vitest run ${wzor}`;
  let kod = 0, out = '';
  try { out = execSync(cmd + ' 2>&1', { encoding: 'utf8', maxBuffer: 1 << 28 }); } catch (e) { kod = e.status; out = e.stdout; }
  writeFileSync(plik, org);
  const pad = (out.match(/^.*(✕|×|FAIL ).*$/gm) ?? []).slice(0, 6).join('\n   ');
  const sum = (out.match(/^Tests:.*$|^\s*Tests\s+.*$/m) ?? [''])[0];
  console.log(`${nazwa}: ${kod === 0 ? 'PRZEŻYŁ' : 'ZABITY'} (kod ${kod}) ${sum.trim()}\n   ${pad}`);
}
