const wynik = [];
for (const [n, plik, a, b, grupa] of M) {
  if (filtr && !filtr.test(n)) continue;
  const f = `${wt}/${plik}`, orig = readFileSync(f, 'utf8');
  if (!orig.includes(a)) { wynik.push(`${n}: WZORZEC NIE ZNALEZIONY`); continue; }
  writeFileSync(f, orig.replace(a, b));
  const pad = [];
  try {
    for (const [cwd, cmd] of T[grupa]) {
      try { execSync(cmd, { cwd: `${wt}/${cwd}`, stdio: 'pipe', timeout: 600000 }); }
      catch (e) {
        const out = String(e.stdout) + String(e.stderr);
        const linie = [...new Set(out.split('\n').filter((l) => /^\s*(×|✕|FAIL|BŁĄD|Error:)/.test(l)).map((l) => l.trim().slice(0, 140)))].slice(0, 2);
        pad.push(`${cwd}: kod ${e.status} ${linie.join(' | ')}`);
      }
    }
  } finally { writeFileSync(f, orig); }
  wynik.push(`${n} [${grupa}]: ${pad.length ? 'ZABITY — ' + pad.join(' ; ') : 'PRZEŻYŁ (zielony)'}`);
  console.error(wynik.at(-1));
}
console.log(wynik.join('\n'));
