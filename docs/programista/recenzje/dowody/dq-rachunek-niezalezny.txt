from fractions import Fraction as F
from functools import lru_cache
def icm(stacks, payouts):
    n=len(stacks); tot=sum(stacks)
    def rec(rem, place, prob, eq):
        if place>=len(payouts): return
        s=sum(stacks[i] for i in rem)
        for i in rem:
            p=prob*F(stacks[i],s); eq[i]+=p*payouts[place]
            rec([j for j in rem if j!=i], place+1, p, eq)
    eq=[F(0)]*n; rec([i for i in range(n) if stacks[i]>0],0,F(1),eq); return eq
def przyklad(nazwa, stacks, payouts, hero, vil, stackBb=10, blind=1):
    eff=min(stacks[hero],stacks[vil]); chip_per_bb=F(eff,stackBb)
    bb=[F(x)/chip_per_bb for x in stacks]
    pula=F(stackBb)+blind          # rywal stawia efektywnie eff (=stackBb bb), ty masz już blind w puli
    doplata=F(stackBb)-blind
    def after(d): return [x+d if j==hero else x-d if j==vil else x for j,x in enumerate(stacks)]
    P=[F(int(p*100),100) for p in payouts]
    w=icm(after(eff),P)[hero]; l=icm(after(-eff),P)[hero]; f=icm(after(-blind*chip_per_bb),P)[hero]
    req=(f-l)/(w-l); chips=doplata/(pula+doplata)
    print(f"{nazwa}: stosy bb={[float(round(x,1)) for x in bb]} pula={float(pula)} doplata={float(doplata)} "
          f"cały stos={stacks[hero]<=stacks[vil]} próg ICM={float(req)*100:.1f}% próg żetony={float(chips)*100:.1f}% (D-890 (S-b)/(2S)={(stackBb-blind)/(2*stackBb)*100:.1f}%) suma nagród={float(sum(P))*100:.0f}%")
