wybrane = sys.argv[2:] or list(M)
for k in wybrane:
    plik, stare, nowe, test = M[k]
    p = wt / plik; src = p.read_text()
    assert src.count(stare) == 1, (k, src.count(stare))
    p.write_text(src.replace(stare, nowe))
    try:
        r = subprocess.run(['pnpm', '--filter', '@szkola/mobile', 'exec', 'jest', '--ci', test], cwd=wt, capture_output=True, text=True)
        out = r.stdout + r.stderr
        linie = [l for l in out.splitlines() if l.strip().startswith(('●', 'Tests:')) and 'Console' not in l]
        print(f'== {k} ({plik}) test {test}: kod {r.returncode} → {"ZABITY" if r.returncode else "PRZEŻYŁ"}')
        for l in dict.fromkeys(linie): print('   ', l.strip())
    finally:
        p.write_text(src)
