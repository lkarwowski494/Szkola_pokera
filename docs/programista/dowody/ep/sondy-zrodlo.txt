// --- ep-sonda.test.ts: pick-hand, strona poprawnej ręki i remisy przy „SDV = 0” (fixture CALLS/FOLDS z ek-pick-hand.test.ts, ziarna 1–300)
function remisy(h, board) { // kombinacje calls ∪ folds (classCombos) bez kart h i stołu; r = remisy (strength równe), w = wygrane h
  const m = new Set([...h, ...board]); const s = strength([...h, ...board]); let r = 0, w = 0, n = 0;
  for (const c of [...CALLS, ...FOLDS].flatMap((x) => classCombos(x))) { if (m.has(c[0]) || m.has(c[1])) continue; n++; const o = strength([...c, ...board]); if (o === s) r++; if (s < o) w++; }
  return { r, w, n };
}
