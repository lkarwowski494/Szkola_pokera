## skrypt
// Sonda 402: czy margines >= 0,01 (warunek inMargin generatora) gwarantuje różne procenty (Intl pl, 0 miejsc)?
const f = new Intl.NumberFormat('pl-PL', { style: 'percent', maximumFractionDigits: 0 });
const req = (pot, bet) => bet / (pot + 2 * bet);
const MIN = Number(process.argv[2] ?? 0.01);
let sprawdzone = 0, kolizje = [];
const udzialy = []; for (let k = 0; k <= 1000; k++) udzialy.push(Math.round(k * 0.001 * 1000) / 1000); // krok 0,001
for (let pot = 1; pot <= 300; pot++) for (let bet = 1; bet <= 3 * pot; bet++) {
  const r = req(pot, bet);
  for (const s of udzialy) {
    const m = Math.abs(s - r);
    if (!(m > 0 && m >= MIN)) continue;
    sprawdzone++;
    if (f.format(s) === f.format(r)) kolizje.push({ pot, bet, s, r, m });
  }
}
console.log(`próg ${MIN}: przypadków w marginesie ${sprawdzone}, kolizji procentów ${kolizje.length}`, kolizje.slice(0, 5));
