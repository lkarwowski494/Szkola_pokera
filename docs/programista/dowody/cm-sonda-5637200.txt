import { join, resolve } from 'node:path';
import { readFileSync } from 'node:fs';
import { it } from 'vitest';
import { equityAfter, generateVariant, type GeneratorTemplate } from '@szkola/poker-core';
import { checkScenarioGenerator, checkScenarioNaEkran, loadScenarioTemplates } from '../../src/scenariusze';
import { compileRanges } from '../../src/ranges';
const root = resolve(import.meta.dirname, '../../../..');
const fxd = join(resolve(import.meta.dirname, '..'), 'fixtures');
const log = (...a: unknown[]) => process.stderr.write('\nSONDA ' + a.join(' '));
it('sonda', { timeout: 60_000 }, () => {
  for (const k of ['cbet-case', 'size-spot', 'icm-call']) for (const { template: t } of loadScenarioTemplates(join(fxd, k))) {
    const g = { id: t.id, stackBb: t.stackBb, positions: t.positions, levels: t.bounds, decisions: t.decisions } as GeneratorTemplate;
    let blad = 'brak';
    try { generateVariant(g, t.levels[0] as never, 1); } catch (e) { blad = String(e).slice(0, 110); }
    log(k, t.id, 'naEkran=', checkScenarioNaEkran(t).length, 'bezCtx:', blad);
  }
  const eq = JSON.parse(readFileSync(join(root, 'tools/equity/equity169.json'), 'utf8'));
  const ctx = { spots: compileRanges(join(root, 'content')).spots, thresholds: { mixedMin: 0.035, mixedLow: 0.25 }, equity169: eq };
  const t = loadScenarioTemplates(join(fxd, 'icm-call'))[0]!.template;
  log('icm BB wobec SB (fixture):', JSON.stringify(checkScenarioGenerator(t as never, 40, 600, ctx)));
  log('icm SB wobec BB:', JSON.stringify(checkScenarioGenerator({ ...t, positions: { hero: 'SB', villain: 'BB' } } as never, 40, 600, ctx)));
  const fold = equityAfter([5200, 3000, 1800], [0.6, 0.4], 2), win = equityAfter([3000, 3000, 4000], [0.6, 0.4], 2);
  log('rachunek: fold', fold, 'win', win, 'wymagane', fold / win);
});
