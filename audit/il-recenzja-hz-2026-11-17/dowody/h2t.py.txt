import sys,re,html
for p in sys.argv[1:]:
    t=open(p,encoding='utf-8',errors='replace').read()
    t=re.sub(r'(?is)<(script|style|noscript)[^>]*>.*?</\1>',' ',t)
    t=re.sub(r'(?s)<[^>]+>',' ',t)
    t=html.unescape(t)
    t=t.replace('’',"'").replace('‘',"'").replace('“','"').replace('”','"').replace('\xa0',' ')
    t=re.sub(r'\s+',' ',t)
    open(p[:-5]+'.txt','w').write(t)
