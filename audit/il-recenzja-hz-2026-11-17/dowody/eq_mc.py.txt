# IL: niezależny rachunek equity (Monte Carlo, własny ewaluator 7 kart, bez equity169 aplikacji).
# Klasa gracza (wszystkie kombinacje offsuit) wobec zakresu rywala (lista klas, równomiernie po kombinacjach, bez kart wspólnych).
import random, sys, itertools
from collections import Counter
R='23456789TJQKA'
def combos(hc):
    a,b=R.index(hc[0]),R.index(hc[1])
    if len(hc)==2: return [((a,s1),(a,s2)) for s1,s2 in itertools.combinations(range(4),2)]
    if hc[2]=='s': return [((a,s),(b,s)) for s in range(4)]
    return [((a,s1),(b,s2)) for s1 in range(4) for s2 in range(4) if s1!=s2]
def rank5(cards):  # najlepszy układ z 7 kart -> krotka porównywalna
    rs=sorted((c[0] for c in cards),reverse=True)
    cnt=Counter(rs); suits=Counter(c[1] for c in cards)
    fl=[s for s,n in suits.items() if n>=5]
    def straight(rset):
        u=sorted(set(rset),reverse=True)
        if 12 in u: u.append(-1)
        run=1
        for i in range(1,len(u)):
            run = run+1 if u[i]==u[i-1]-1 else 1
            if run>=5: return u[i]+4
        return None
    if fl:
        fr=[c[0] for c in cards if c[1]==fl[0]]
        sf=straight(fr)
        if sf is not None: return (8,sf)
    g=sorted(cnt.items(),key=lambda x:(x[1],x[0]),reverse=True)
    if g[0][1]==4: return (7,g[0][0],max(r for r in rs if r!=g[0][0]))
    if g[0][1]==3 and len(g)>1 and g[1][1]>=2: return (6,g[0][0],g[1][0])
    if fl: return (5,)+tuple(sorted((c[0] for c in cards if c[1]==fl[0]),reverse=True)[:5])
    st=straight(rs)
    if st is not None: return (4,st)
    if g[0][1]==3: return (3,g[0][0])+tuple([r for r in rs if r!=g[0][0]][:2])
    if g[0][1]==2 and g[1][1]==2: return (2,g[0][0],g[1][0],max(r for r in rs if r not in (g[0][0],g[1][0])))
    if g[0][1]==2: return (1,g[0][0])+tuple([r for r in rs if r!=g[0][0]][:3])
    return (0,)+tuple(rs[:5])
def eq(hero_cls, rng_cls, n, rnd):
    hc=combos(hero_cls); vc=[c for k in rng_cls for c in combos(k)]
    deck=[(r,s) for r in range(13) for s in range(4)]
    w=0.0; t=0
    while t<n:
        h=rnd.choice(hc); v=rnd.choice(vc)
        if set(h)&set(v): continue
        rest=[c for c in deck if c not in h and c not in v]
        b=rnd.sample(rest,5)
        a=rank5(list(h)+b); c=rank5(list(v)+b)
        w+= 1 if a>c else 0.5 if a==c else 0; t+=1
    return w/n

        for hcls in hands:
