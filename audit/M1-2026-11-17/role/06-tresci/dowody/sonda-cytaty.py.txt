# audyt 06: weryfikacja dosłowna cytatów rejestru na surowym tekście źródeł (curl + html->tekst / pdftotext)
# użycie: python3 -I cytaty.py <rejestr.json> <katalog-cache> ID...
import sys, json, re, html, os, subprocess, hashlib, unicodedata
rej = {t['id']: t for t in json.load(open(sys.argv[1], encoding='utf8'))}
cache = sys.argv[2]; os.makedirs(cache, exist_ok=True)
def norm(s):
    s = unicodedata.normalize('NFKC', s)
    for a, b in [('’', "'"), ('‘', "'"), ('“', '"'), ('”', '"'), ('–', '-'), ('—', '-'), ('−', '-'), (' ', ' ')]:
        s = s.replace(a, b)
    return re.sub(r'\s+', ' ', s).strip().lower()
def text(url):
    h = hashlib.sha1(url.encode()).hexdigest()[:12]; f = os.path.join(cache, h)
    if not os.path.exists(f + '.raw'):
        r = subprocess.run(['curl', '-sSL', '--max-time', '45', '-A', 'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 Chrome/124 Safari/537.36', '-o', f + '.raw', '-w', '%{http_code}', url], capture_output=True, text=True)
        open(f + '.code', 'w').write(r.stdout or 'ERR ' + r.stderr[:100])
    code = open(f + '.code').read()
    raw = open(f + '.raw', 'rb').read() if os.path.exists(f + '.raw') else b''
    if raw[:5] == b'%PDF-':
        t = subprocess.run(['pdftotext', '-layout', f + '.raw', '-'], capture_output=True, text=True).stdout
    else:
        t = raw.decode('utf8', 'ignore'); t = re.sub(r'<script.*?</script>|<style.*?</style>', ' ', t, flags=re.S | re.I); t = html.unescape(re.sub(r'<[^>]+>', ' ', t))
    return code, norm(t)
for id in sys.argv[3:]:
    t = rej[id]
    for z in t.get('zrodla', []):
        a = z.get('adres', '')
        if not a.startswith('http'): continue
        url = a.split(' ')[0]
        code, tx = text(url)
        q = z.get('cytat') or ''
        parts = [p for p in re.split(r'\s*(?:\[…\]|…|\.\.\.|\|)\s*', q) if len(p.strip()) >= 8]
        ok = [norm(p.strip(' "„”')) in tx for p in parts]
        st = 'BRAK-CYTATU' if not parts else ('ZNALEZIONY' if all(ok) else ('CZĘŚCIOWO' if any(ok) else 'NIE-ZNALEZIONY'))
        print(f"{id}\tHTTP {code}\tlen {len(tx)}\t{st}\t{url}\t„{q[:160]}”")
