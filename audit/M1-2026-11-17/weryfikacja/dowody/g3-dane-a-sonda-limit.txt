    const strefa = 'America/New_York';
    const zegar = (iso: string) => ({ teraz: () => Date.parse(iso), strefa: () => strefa, dzienLokalny: (ms: number) => new Date(ms - 5 * 3600e3).toISOString().slice(0, 10) });
    const L = KONFIGURACJA.LIMIT_AKTYWNOSCI_DZIENNIE;
    console.log('LIMIT_AKTYWNOSCI_DZIENNIE =', L);
    // wieczór dnia 1: odczyt przy fokusie zwrócił zuzyte = L
    console.log('dzień 1, 23:00, zuzyte (odczyt przy fokusie) =', L, '→', JSON.stringify(limitZeZuzycia(L, { pro: false, zegar: zegar('2026-11-18T04:00:00Z') })));
    // rano dnia 2 bez nowego odczytu: ekran ma nadal zuzyte = L (stan z useOdczytPostepu), render liczy limit na nowym dniu
    console.log('dzień 2, 08:00, ten sam stan ekranu (zuzyte =', L + ') →', JSON.stringify(limitZeZuzycia(L, { pro: false, zegar: zegar('2026-11-18T13:00:00Z') })));
    // to samo po nowym odczycie (zuzyteDzis dla nowego dnia = 0), jak przy wejściu na ekran albo w przygotujStart
    console.log('dzień 2, 08:00, po nowym odczycie (zuzyte = 0) →', JSON.stringify(limitZeZuzycia(0, { pro: false, zegar: zegar('2026-11-18T13:00:00Z') })));
