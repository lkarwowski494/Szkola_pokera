# kod parsera (szer.mjs):
import { readFileSync } from 'node:fs';
const [plik, tekst, rozm] = process.argv.slice(2);
const b = readFileSync(plik); const dv = new DataView(b.buffer, b.byteOffset, b.byteLength);
const n = dv.getUint16(4); const T = {};
for (let i = 0; i < n; i++) { const o = 12 + 16 * i; T[b.toString('latin1', o, o + 4)] = dv.getUint32(o + 8); }
const upm = dv.getUint16(T.head + 18); const nH = dv.getUint16(T.hhea + 34);
const cm = T.cmap; const nt = dv.getUint16(cm + 2); let sub;
for (let i = 0; i < nt; i++) { const p = dv.getUint16(cm + 4 + 8 * i), e = dv.getUint16(cm + 6 + 8 * i), off = dv.getUint32(cm + 8 + 8 * i); if (p === 3 && e === 1) sub = cm + off; }
const segX2 = dv.getUint16(sub + 6), sc = segX2 / 2; const ends = sub + 14, starts = ends + segX2 + 2, deltas = starts + segX2, ros = deltas + segX2;
const gid = (c) => { for (let i = 0; i < sc; i++) { const e = dv.getUint16(ends + 2 * i); if (c <= e) { const s = dv.getUint16(starts + 2 * i); if (c < s) return 0; const d = dv.getInt16(deltas + 2 * i), r = dv.getUint16(ros + 2 * i); if (!r) return (c + d) & 0xffff; const g = dv.getUint16(ros + 2 * i + r + 2 * (c - s)); return g ? (g + d) & 0xffff : 0; } } return 0; };
const adv = (g) => dv.getUint16(T.hmtx + 4 * Math.min(g, nH - 1));
let s = 0; for (const ch of tekst) s += adv(gid(ch.codePointAt(0)));
console.log(`${tekst}: ${(s / upm * Number(rozm)).toFixed(1)} pt przy ${rozm} pt (upm ${upm}, bez kerningu)`);
