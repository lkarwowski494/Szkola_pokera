# Niezależny pomiar: kontrast kropki muted na polu felt1*0.85+surface*0.15 (RangeGrid.tsx:30,34,42) i udział pól >=0.95 w bazie pl
import re, sys, json, sqlite3
tok = open(sys.argv[1], encoding='utf-8').read()
def blok(n):
    m = re.search(r'const %s(?:: Paleta)? = \{(.*?)\n\};' % n, tok, re.S); return dict(re.findall(r"(\w+): '(#[0-9A-Fa-f]{6})'", m.group(1)))
L = blok('light'); D = {**L, **blok('dark')}; LH = {**L, **blok('lightHigh')}; DH = {**D, **blok('darkHigh')}
def rgb(h): return [int(h[i:i+2], 16) for i in (1, 3, 5)]
def lum(c):
    f = lambda v: (v/255)/12.92 if v/255 <= 0.04045 else ((v/255+0.055)/1.055)**2.4
    r, g, b = map(f, c); return 0.2126*r+0.7152*g+0.0722*b
def cr(a, b):
    x, y = sorted([lum(a), lum(b)], reverse=True); return (x+0.05)/(y+0.05)
for nazwa, P in [('light', L), ('lightHigh', LH), ('dark', D), ('darkHigh', DH)]:
    pole = [0.85*f + 0.15*s for f, s in zip(rgb(P['felt1']), rgb(P['surface']))]
    print(f"{nazwa}: felt1={P['felt1']} surface={P['surface']} muted={P['muted']} ink={P['ink']} pole=#{''.join('%02X'%round(v) for v in pole)} | kropka muted/pole {cr(rgb(P['muted']), pole):.2f}:1 | muted/surface {cr(rgb(P['muted']), rgb(P['surface'])):.2f}:1 | ink/pole {cr(rgb(P['ink']), pole):.2f}:1 | {'BRAK' if cr(rgb(P['muted']), pole) < 3 else 'OK'} wobec 3:1")
